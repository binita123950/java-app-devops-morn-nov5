pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/binita123950/java-app-devops-morn-nov5.git'
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t jenkins-demo .'
            }
        }

        stage('Run Container') {
            steps {
                sh '''
                docker stop jenkins-demo || true
                docker rm jenkins-demo || true
                docker run -d -p 3000:3000 --name jenkins-demo jenkins-demo
                '''
            }
        }
    }
}
